    <script>
        let transactions = JSON.parse(localStorage.getItem('infaq_txs')) || [];
        let isPublicMode = false;

        function initApp() {
            loadOfficerNames();
            renderData();
            document.getElementById('tx-date').valueAsDate = new Date();
            document.getElementById('print-date').innerText = new Date().toLocaleDateString('id-ID', { day: 'numeric', month: 'long', year: 'numeric' });
        }

        function saveOfficerNames() {
            const kepala = document.getElementById('input-kepala').value;
            const bendahara = document.getElementById('input-bendahara').value;
            localStorage.setItem('officer_kepala', kepala);
            localStorage.setItem('officer_bendahara', bendahara);
            loadOfficerNames();
            alert('Nama pejabat berhasil disimpan!');
        }

        function loadOfficerNames() {
            const kepala = localStorage.getItem('officer_kepala') || '';
            const bendahara = localStorage.getItem('officer_bendahara') || '';
            document.getElementById('input-kepala').value = kepala;
            document.getElementById('input-bendahara').value = bendahara;
            document.getElementById('print-kepala').innerText = kepala || '(...................................)';
            document.getElementById('print-bendahara').innerText = bendahara || '(...................................)';
        }

        function addTransaction(e) {
            e.preventDefault();
            const newTx = {
                id: Date.now(),
                date: document.getElementById('tx-date').value,
                type: document.getElementById('tx-type').value,
                category: document.getElementById('tx-category').value,
                description: document.getElementById('tx-desc').value,
                amount: parseFloat(document.getElementById('tx-amount').value)
            };
            transactions.push(newTx);
            localStorage.setItem('infaq_txs', JSON.stringify(transactions));
            document.getElementById('form-transaction').reset();
            document.getElementById('tx-date').valueAsDate = new Date();
            renderData();
        }

        function deleteTransaction(id) {
            if (confirm('Apakah Anda yakin ingin menghapus transaksi ini?')) {
                transactions = transactions.filter(t => t.id !== id);
                localStorage.setItem('infaq_txs', JSON.stringify(transactions));
                renderData();
            }
        }

        function renderData() {
            const tbody = document.getElementById('table-body');
            tbody.innerHTML = '';
            let totalMasuk = 0, totalKeluar = 0;

            transactions.sort((a,b) => new Date(b.date) - new Date(a.date)).forEach(tx => {
                if (tx.type === 'masuk') totalMasuk += tx.amount;
                else totalKeluar += tx.amount;

                const tr = document.createElement('tr');
                tr.className = 'border-b hover:bg-slate-50';
                tr.innerHTML = `
                    <td class="p-2">${tx.date}</td>
                    <td class="p-2"><span class="px-2 py-0.5 rounded text-xs font-bold ${tx.type === 'masuk' ? 'bg-emerald-100 text-emerald-800' : 'bg-rose-100 text-rose-800'}">${tx.type.toUpperCase()}</span></td>
                    <td class="p-2">${tx.category}</td>
                    <td class="p-2">${tx.description}</td>
                    <td class="p-2 text-right font-semibold ${tx.type === 'masuk' ? 'text-emerald-600' : 'text-rose-600'}">Rp ${tx.amount.toLocaleString('id-ID')}</td>
                    <td class="p-2 text-center no-print">
                        ${!isPublicMode ? `<button onclick="deleteTransaction(${tx.id})" class="text-rose-500 hover:text-rose-700"><i class="fas fa-trash"></i></button>` : '-'}
                    </td>
                `;
                tbody.appendChild(tr);
            });

            document.getElementById('txt-pemasukan').innerText = `Rp ${totalMasuk.toLocaleString('id-ID')}`;
            document.getElementById('txt-pengeluaran').innerText = `Rp ${totalKeluar.toLocaleString('id-ID')}`;
            document.getElementById('txt-saldo').innerText = `Rp ${(totalMasuk - totalKeluar).toLocaleString('id-ID')}`;
        }

        function toggleMode() {
            isPublicMode = !isPublicMode;
            document.getElementById('section-form').style.display = isPublicMode ? 'none' : 'block';
            document.getElementById('section-officers').style.display = isPublicMode ? 'none' : 'block';
            document.getElementById('btn-mode').innerHTML = isPublicMode 
                ? '<i class="fas fa-user-shield"></i> Mode Pengurus' 
                : '<i class="fas fa-eye"></i> Mode Publik (Read-Only)';
            renderData();
        }

        window.onload = initApp;
    </script>
